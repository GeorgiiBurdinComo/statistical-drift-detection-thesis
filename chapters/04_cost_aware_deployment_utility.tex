\chapter{Cost-aware Deployment Utility}
\label{ch:cost-deployment-utility}

This chapter formalises deployment selection by combining inference cost with asymmetric misclassification penalties. If business-error costs can be specified, the result is a per-request expected cost that gives a single scalar ranking of candidate models. If such costs cannot be stated reliably, the same framework can still be used as a parametric decision rule through break-even analysis and constraint-based selection. The chapter builds on the metrics and candidate-comparison setup in Chapter~\ref{ch:drift} (Sections~\ref{sec:classification-metrics} and~\ref{sec:model-comparison-manual}).

\section{Cost components}

Each request to model $m$ at evaluation time $t$ incurs two types of cost:
\begin{enumerate}
    \item Inference cost $C_t(m)$: the API charge in USD per request, observed from logged usage and pricing (in the same units as Section~\ref{sec:model-comparison-manual}).
    \item Business error cost: a penalty determined by the prediction outcome. Correct predictions (TP, TN) add no further business cost. Misclassifications incur:
    \begin{itemize}
        \item $C_{\mathrm{FP}}$: cost of a false positive (for example, an irrelevant post marked as relevant and sent to unnecessary downstream processing),
        \item $C_{\mathrm{FN}}$: cost of a false negative (for example, a relevant post missed by the system, creating a gap in campaign coverage).
    \end{itemize}
\end{enumerate}

\section{Derivation of per-request expected cost}

\paragraph{Step 1: confusion-matrix decomposition.}
On an evaluation batch $\mathcal{D}_n$ of $n$ labelled items, model $m$ at time $t$ produces confusion counts $\mathrm{TP}_t(m)$, $\mathrm{FP}_t(m)$, $\mathrm{TN}_t(m)$, and $\mathrm{FN}_t(m)$ satisfying
\[
\mathrm{TP}_t(m)+\mathrm{FP}_t(m)+\mathrm{TN}_t(m)+\mathrm{FN}_t(m)=n.
\]

\paragraph{Step 2: total batch cost.}
The total batch cost is the sum of inference cost and business error cost:
\begin{equation}
    \mathrm{TotalCost}_t(m)
    = \underbrace{n \cdot C_t(m)}_{\text{inference}}
    + \underbrace{C_{\mathrm{FP}}\cdot\mathrm{FP}_t(m)
    + C_{\mathrm{FN}}\cdot\mathrm{FN}_t(m)}_{\text{business error cost}}.
\end{equation}

\paragraph{Step 3: per-request expected cost.}
Dividing by $n$ gives the expected cost per request:
\begin{equation}
\label{eq:expected-cost}
    E_t[\mathcal{C}](m)
    = C_t(m)
    + \frac{C_{\mathrm{FP}}\cdot\mathrm{FP}_t(m)
          + C_{\mathrm{FN}}\cdot\mathrm{FN}_t(m)}{n}.
\end{equation}

\paragraph{Step 4: model selection rule under fixed costs.}
When $C_{\mathrm{FP}}$ and $C_{\mathrm{FN}}$ are available, the preferred deployment model is the one that minimises expected cost:
\begin{equation}
\label{eq:model-selection}
    m_t^\star
    \;\in\;
    \arg\min_{m\in\mathcal{M}_{\mathrm{cand}}}\;
    E_t[\mathcal{C}](m).
\end{equation}

\section{Rate form}

Let
\[
\hat{\pi}_t=\frac{\mathrm{TP}_t(m)+\mathrm{FN}_t(m)}{n}
\]
denote the empirical positive prevalence on the evaluation set at time $t$. Also define the empirical false-positive rate and false-negative rate of model $m$ at time $t$ by
\[
\widehat{\mathrm{FPR}}_t(m)
=
\frac{\mathrm{FP}_t(m)}{\mathrm{FP}_t(m)+\mathrm{TN}_t(m)},
\qquad
\widehat{\mathrm{FNR}}_t(m)
=
\frac{\mathrm{FN}_t(m)}{\mathrm{FN}_t(m)+\mathrm{TP}_t(m)},
\]
whenever the denominators are nonzero.

Then
\[
    \frac{\mathrm{FP}_t(m)}{n}
    = \widehat{\mathrm{FPR}}_t(m)\,(1-\hat{\pi}_t),
    \qquad
    \frac{\mathrm{FN}_t(m)}{n}
    = \widehat{\mathrm{FNR}}_t(m)\,\hat{\pi}_t,
\]
so~\eqref{eq:expected-cost} can be written as
\begin{equation}
\label{eq:expected-cost-rate}
    E_t[\mathcal{C}](m)
    = C_t(m)
    + C_{\mathrm{FP}}\,\widehat{\mathrm{FPR}}_t(m)\,(1-\hat{\pi}_t)
    + C_{\mathrm{FN}}\,\widehat{\mathrm{FNR}}_t(m)\,\hat{\pi}_t.
\end{equation}

This form makes the weighting explicit: the false-positive penalty is weighted by the empirical negative prevalence $1-\hat{\pi}_t$, and the false-negative penalty by the empirical positive prevalence $\hat{\pi}_t$.

\section{Cost parameters under uncertainty}
\label{sec:cost-parameters-uncertainty}

In practice, it may be impossible to assign reliable monetary values to $C_{\mathrm{FP}}$ and $C_{\mathrm{FN}}$. In that case, the expected-cost utility is treated not as a single point estimate but as a parametric decision rule. Reparameterise the cost pair as
\[
C_{\mathrm{FP}}=\lambda R,
\qquad
C_{\mathrm{FN}}=\lambda,
\qquad
R=\frac{C_{\mathrm{FP}}}{C_{\mathrm{FN}}},
\qquad
\lambda=C_{\mathrm{FN}}>0,
\]
so that $R$ carries the relative FP--FN trade-off and $\lambda$ the overall scale of business-error severity relative to inference cost. This is a change of variables only, not an additional assumption.

Substituting into~\eqref{eq:expected-cost} gives
\begin{equation}
\label{eq:expected-cost-lambda-R}
    E_t[\mathcal{C}](m)
    = C_t(m)
    + \lambda\cdot\frac{R\,\mathrm{FP}_t(m)+\mathrm{FN}_t(m)}{n}.
\end{equation}
Equivalently, in rate form,
\begin{equation}
\label{eq:expected-cost-rate-lambda-R}
    E_t[\mathcal{C}](m)
    = C_t(m)
    + \lambda\Bigl(
        R\,\widehat{\mathrm{FPR}}_t(m)\,(1-\hat{\pi}_t)
        + \widehat{\mathrm{FNR}}_t(m)\,\hat{\pi}_t
      \Bigr).
\end{equation}

\section{Pairwise break-even analysis}
\label{sec:break-even-analysis}

For two candidate models $m_i,m_j\in\mathcal{M}_{\mathrm{cand}}$, define
\[
\Delta C_t^{ij}=C_t(m_i)-C_t(m_j),\qquad
\Delta \mathrm{FP}_t^{ij}=\mathrm{FP}_t(m_i)-\mathrm{FP}_t(m_j),\qquad
\Delta \mathrm{FN}_t^{ij}=\mathrm{FN}_t(m_i)-\mathrm{FN}_t(m_j).
\]
Then $m_i$ is preferred to $m_j$ under~\eqref{eq:expected-cost} if and only if
\[
\Delta C_t^{ij}
+
\frac{
C_{\mathrm{FP}}\Delta \mathrm{FP}_t^{ij}
+
C_{\mathrm{FN}}\Delta \mathrm{FN}_t^{ij}
}{n}
\le 0.
\]

The break-even boundary between the two models is therefore
\begin{equation}
\label{eq:break-even-count}
    \Delta C_t^{ij}
    +
    \frac{
    C_{\mathrm{FP}}\Delta \mathrm{FP}_t^{ij}
    +
    C_{\mathrm{FN}}\Delta \mathrm{FN}_t^{ij}
    }{n}
    =0.
\end{equation}
Under the parameterisation $C_{\mathrm{FP}}=\lambda R$ and $C_{\mathrm{FN}}=\lambda$, this becomes
\begin{equation}
\label{eq:break-even-lambda-R}
    \Delta C_t^{ij}
    +
    \lambda\cdot
    \frac{
    R\,\Delta \mathrm{FP}_t^{ij}
    +
    \Delta \mathrm{FN}_t^{ij}
    }{n}
    =0.
\end{equation}

For fixed $R$, the break-even severity level is
\begin{equation}
\label{eq:lambda-breakpoint}
    \lambda_{ij}^\star(R)
    =
    -\frac{n\,\Delta C_t^{ij}}
    {R\,\Delta \mathrm{FP}_t^{ij}+\Delta \mathrm{FN}_t^{ij}},
\end{equation}
provided the denominator is nonzero. For fixed $\lambda$, the break-even trade-off ratio is
\begin{equation}
\label{eq:R-breakpoint}
    R_{ij}^\star(\lambda)
    =
    -\frac{n\,\Delta C_t^{ij}/\lambda+\Delta \mathrm{FN}_t^{ij}}
    {\Delta \mathrm{FP}_t^{ij}},
\end{equation}
provided $\Delta \mathrm{FP}_t^{ij}\neq 0$.

These breakpoints identify parameter values at which deployment preference switches between the two candidates. Across all candidate pairs, the resulting boundaries partition the admissible $(R,\lambda)$ plane into regions within which the minimiser of~\eqref{eq:expected-cost-lambda-R} is constant; Figure~\ref{fig:cost-decision-map} draws that partition for the candidates of this study. When the relevant denominator is zero, there is no unique finite breakpoint in that one-dimensional slice. When the solution lies outside the admissible region $(R\ge 0,\lambda>0)$, no operational switch occurs under nonnegative cost assumptions.

\paragraph{Dominance.}
If, for some pair $(m_i,m_j)$,
\[
C_t(m_i)\le C_t(m_j),\qquad
\mathrm{FP}_t(m_i)\le \mathrm{FP}_t(m_j),\qquad
\mathrm{FN}_t(m_i)\le \mathrm{FN}_t(m_j),
\]
with at least one strict inequality, then $m_i$ dominates $m_j$ for all $C_{\mathrm{FP}},C_{\mathrm{FN}}\ge 0$. Such dominated candidates can be removed before sensitivity analysis.

\section{Constraint-based selection without monetary elicitation}
\label{sec:constraint-based-selection}

If reliable monetary values cannot be assigned to $C_{\mathrm{FP}}$ and $C_{\mathrm{FN}}$, model selection can instead be formulated as constrained optimisation over operational tolerances. Let $\mathcal{F}_t\subseteq\mathcal{M}_{\mathrm{cand}}$ denote the feasible set under stated feasibility constraints, and let $J_t(m)$ denote a secondary objective over feasible candidates. The deployment rule becomes
\[
m_t^\star\in\arg\min_{m\in\mathcal{F}_t} J_t(m),
\]
with the exact form determined by the operational requirement.

For completeness, define the empirical recall and precision of model $m$ at time $t$ by
\[
\widehat{\mathrm{Recall}}_t(m)
=
\frac{\mathrm{TP}_t(m)}{\mathrm{TP}_t(m)+\mathrm{FN}_t(m)},
\qquad
\widehat{\mathrm{Precision}}_t(m)
=
\frac{\mathrm{TP}_t(m)}{\mathrm{TP}_t(m)+\mathrm{FP}_t(m)},
\]
whenever the denominators are nonzero.
\[
\widehat{\mathrm{FNR}}_t(m)
=
\frac{\mathrm{FN}_t(m)}{\mathrm{FN}_t(m)+\mathrm{TP}_t(m)}
=
1-\widehat{\mathrm{Recall}}_t(m).
\]

\paragraph{Recall-constrained cost minimisation.}
If the business can specify a minimum acceptable recall $\tau_{\mathrm{rec}}$, define
\[
\mathcal{F}_t^{\mathrm{rec}}(\tau_{\mathrm{rec}})
=
\left\{
m\in\mathcal{M}_{\mathrm{cand}}:
\widehat{\mathrm{Recall}}_t(m)\ge\tau_{\mathrm{rec}}
\right\}.
\]
Then select
\begin{equation}
\label{eq:recall-constrained-selection}
    m_t^\star(\tau_{\mathrm{rec}})
    \in
    \arg\min_{m\in\mathcal{F}_t^{\mathrm{rec}}(\tau_{\mathrm{rec}})}
    C_t(m).
\end{equation}
Equivalently,
\[
\widehat{\mathrm{FNR}}_t(m)\le 1-\tau_{\mathrm{rec}}.
\]

\paragraph{Budget-constrained miss-rate minimisation.}
If the business can specify a per-request inference budget $B$, define
\[
\mathcal{F}_t^{\mathrm{bud}}(B)
=
\left\{
m\in\mathcal{M}_{\mathrm{cand}}:
C_t(m)\le B
\right\}.
\]
Then select
\begin{equation}
\label{eq:budget-constrained-selection}
    m_t^\star(B)
    \in
    \arg\min_{m\in\mathcal{F}_t^{\mathrm{bud}}(B)}
    \widehat{\mathrm{FNR}}_t(m),
\end{equation}
or, equivalently,
\[
m_t^\star(B)
\in
\arg\max_{m\in\mathcal{F}_t^{\mathrm{bud}}(B)}
\widehat{\mathrm{Recall}}_t(m).
\]

\paragraph{Precision-constrained recall maximisation.}
If unnecessary downstream processing must be limited, let $\tau_{\mathrm{prec}}$ denote a minimum acceptable precision. Define
\[
\mathcal{F}_t^{\mathrm{prec}}(\tau_{\mathrm{prec}})
=
\left\{
m\in\mathcal{M}_{\mathrm{cand}}:
\widehat{\mathrm{Precision}}_t(m)\ge\tau_{\mathrm{prec}}
\right\}.
\]
Then select
\begin{equation}
\label{eq:precision-constrained-selection}
    m_t^\star(\tau_{\mathrm{prec}})
    \in
    \arg\max_{m\in\mathcal{F}_t^{\mathrm{prec}}(\tau_{\mathrm{prec}})}
    \widehat{\mathrm{Recall}}_t(m).
\end{equation}

\section{Remark on prevalence and evaluation scope}

Full-benchmark prevalence is $\hat{\pi}\approx0.569$ ($682$ positive of $1199$); on $\mathcal{D}_n$ it is $\hat{\pi}\approx0.833$ ($250$ of $300$). By Section~\ref{sec:selection-bias}, expected cost, break-even, and constrained selection therefore use $\mathcal{D}_N$ so that~\eqref{eq:expected-cost-rate} is weighted by operational prevalence. Rescaling panel FP/FN by $N/n$ does not fix this: $\hat{\pi}_t$ stays at the panel value. After a McNemar alert (Section~\ref{sec:mcnemar-section}), candidates are ranked by $E_t[\mathcal{C}]$ or by the constrained rules above.
